// Compute the total home ownership cost for multiple users.
// A mortgage payment of 0 ends the program.
start
   Declarations
      num mortgagePayment
      num utilities
      num taxes
      num upkeep
      num total

   startUp()
   while mortgagePayment <> 0
      mainLoop()
   endwhile
   finishUp()
stop

startUp()
   output "Enter your mortgage payment or 0 to quit"
   input mortgagePayment
return

mainLoop()
   output "Enter utilities"
   input utilities
   output "Enter taxes"
   input taxes
   output "Enter amount for upkeep"
   input upkeep
   total = mortgagePayment + utilities + taxes + upkeep
   output "Total is ", total
   output "Enter your mortgage payment or 0 to quit"
   input mortgagePayment
return

finishUp()
   output "End of program"
return

// Tests: mortgage 1000, utilities 200, taxes 150, upkeep 50 => 1400;
// mortgage 0 => end.
