// Determine whether the average of two tests meets the passing score.
// Enter 0 for the first score to stop processing students.
start
   Declarations
      num firstTest
      num secondTest
      num average
      num PASSING = 60

   output "Enter first score or 0 to quit"
   input firstTest
   while firstTest <> 0
      output "Enter second score"
      input secondTest
      average = (firstTest + secondTest) / 2
      output "Average is ", average
      if average >= PASSING then
         output "Pass"
      else
         output "Fail"
      endif
      output "Enter first score or 0 to quit"
      input firstTest
   endwhile
stop

// Tests: 100, 80 => 90 Pass; 50, 60 => 55 Fail;
// 60, 60 => 60 Pass; first score 0 => stop.
